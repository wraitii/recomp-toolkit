# The GPU interface, its test double, and the platform backend.
add_library(gpu_fake OBJECT fake/fake_device.cpp)
target_include_directories(gpu_fake PUBLIC ${RECOMP_ROOT} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(gpu_fake PRIVATE ${RECOMP_WARN_STRICT})
recomp_optimize(gpu_fake 1)

if(NOT IOS AND NOT ANDROID)
add_executable(gpu_fake_tests fake/tests/gpu_fake_tests.cpp)
target_link_libraries(gpu_fake_tests PRIVATE gpu_fake)
target_compile_options(gpu_fake_tests PRIVATE ${RECOMP_WARN_STRICT})
recomp_optimize(gpu_fake_tests 1)
recomp_test_binary(gpu_fake_tests)
add_test(NAME gpu_fake_tests COMMAND gpu_fake_tests WORKING_DIRECTORY ${RECOMP_ROOT})
set_tests_properties(gpu_fake_tests PROPERTIES LABELS nogame)
endif()

add_library(gpu_backend STATIC wgpu_device.cpp)
if(APPLE)
  target_sources(gpu_backend PRIVATE wgpu_surface.mm)
  target_compile_options(gpu_backend PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  target_link_libraries(gpu_backend PUBLIC "-framework QuartzCore" "-framework Foundation")
else()
  target_sources(gpu_backend PRIVATE wgpu_surface.cpp)
endif()
set_target_properties(gpu_backend PROPERTIES ARCHIVE_OUTPUT_DIRECTORY ${RECOMP_ARCHIVE_DIR})
target_include_directories(gpu_backend PUBLIC ${RECOMP_ROOT} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(gpu_backend PRIVATE ${RECOMP_WARN_HOST})
target_link_libraries(gpu_backend PUBLIC recomp_platform d3d8_wgpu)
recomp_link_sdl(gpu_backend)
recomp_optimize(gpu_backend 1)

if(NOT IOS AND NOT ANDROID)
add_executable(gpu_wgpu_tests tests/gpu_contract_tests.cpp)
target_link_libraries(gpu_wgpu_tests PRIVATE gpu_backend)
target_compile_options(gpu_wgpu_tests PRIVATE ${RECOMP_WARN_STRICT})
recomp_optimize(gpu_wgpu_tests 1)
recomp_test_binary(gpu_wgpu_tests)
add_test(NAME gpu_wgpu_tests COMMAND gpu_wgpu_tests WORKING_DIRECTORY ${RECOMP_ROOT})
set_tests_properties(gpu_wgpu_tests PROPERTIES LABELS gpu)
endif()
